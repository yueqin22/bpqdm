# Forward to the real headers (the gen_dir forwarding headers sit at the front
# of the include path, ahead of every shim).
#
# MotorPmsmParEst.c only includes MotorInclude.h, but that chain reaches the
# whole motor include tree and, through SystemDefine.h -> DSP2803x_Device.h, the
# chip's peripheral register model.
#
# The peripheral headers below are pure `struct` + `extern` declarations and
# compile under MSVC, so they are forwarded verbatim: the register structs stay
# the chip's, which is what makes "did the module write AQCSFRC / DBCTL / TBPRD"
# comparable at all. The module touches exactly three peripherals:
#   EPwm1/2/3Regs  SynInitPosDetSetTs (TBPRD / CMPB) and SynInitPosDetSetPwm
#                  (AQCSFRC / DBCTL / AQCTLA / AQCTLB);
#   GpioDataRegs   the DisableDrive() / EnableDrive() macros of SystemDefine.h,
#                  in SynTuneInitPos, SynInitPosDetect, SynTunePGZero_* and
#                  SynTuneBemf;
#   EQepRegs       QDECCTL (SynTunePGZero_No_Load case 1 and case 2) and
#                  QPOSCNT via GetQepCnt().
# EQepRegs is special: MotorEncoder.h declares it as a *pointer*
# (`extern struct EQEP_REGS *EQepRegs;`), so the case mock defines a real
# `struct EQEP_REGS` instance and points EQepRegs at it, and the driver puts
# that instance in the compared shared image.
#
# Deliberately NOT forwarded (host_patch.json supplies a patched whole copy):
#   DSP2803x_Device.h  -- `extern cregister volatile unsigned int IFR;` and the
#                         EINT/DINT/EALLOW/EDIS `asm(...)` macros are C28x-only.
#                         Replaced by cases/MotorPmsmParEst/shim/DSP2803x_Device.h.
#   DSP2803x_PieVect.h -- `typedef interrupt void(*PINT)(void);`
#   MotorInclude.h     -- four `extern interrupt void ..._isr(void);` prototypes
#   SubPrgInclude.h    -- `(long int)` prototypes, illegal once the shim folds
#                         `int` to the target's 16 bits (MSVC C2632)
#
# MotorInvProtectInclude.h / MotorEncoder.h / MotorVCInclude.h and friends also
# quote-include MotorInclude.h, but they need no patch: the top-level include
# from MotorPmsmParEst.c already resolved to the patched gen_dir copy, whose
# guard blocks the real file no matter which directory the quoted include is
# resolved from.
#
# Not forwarded either, for the same reason MotorInvProtect / MotorMain do not
# forward them: DSP2803x_Examples.h / DSP2803x_GlobalPrototypes.h /
# DSP2803x_DefaultISR.h are full of `__interrupt` and `asm()`. DSP_CLOCK comes
# from SystemDefine.h itself (it `#define`s DSP_CLOCK60 and then DSP_CLOCK).
