# f_menu.h / f_ui.h / f_p2p.h / f_comm.h。这几个都必须用真的:
#   - f_frqSrc.h  提供本模块自己的 extern 声明 (frq / frqAim / maxFrq /
#                 upperFrq / lowerFrq / frqFlag / frqXy / jumpFrq1 / jumpFrq2 ...)
#                 以及 struct FRQ_XY / union FRQ_FLAG / JUMP_FRQ 的布局;

#                 ABS_INT16 / ABS_INT32 —— 本模块的绝对值与周期常量全靠它们;
#   - f_runSrc.h  提供 union RUN_FLAG / union RUN_CMD 的 extern、
#                 accDecFrqPrcFlag、runDirPanelOld、RUN_MODE_* 与 ACC_DEC_FRQ_*;
#   - f_io.h      提供 aiDeal[] / struct AI_DEAL / AI_NUMBER / pulseInSet /
#                 struct DI_FUNC / struct DI_SELECT_FUNC / diFunc / diSelectFunc;
#   - f_menu.h    提供 menu0DispStatus / keyFunc / GetNumberDigit1 的声明;
#   - f_ui.h      提供 frqAimDisp / frqXDisp / frqYDisp / frqAimPLCDisp / runSrc;
#   - f_p2p.h     提供 struct P2P_DATA 与 p2pData (P2P_REV_FRQ_SET 分支要用);
#   - f_comm.h    提供 dpFrqAim / commProtocolSec / PROFIBUS 常量。
#
# f_funcCode.h 由 f_runSrc.h 拉进来, 提供 FUNCCODE_ALL funcCode 与全部
# FUNCCODE_frqCalcSrc_* / FUNCCODE_frqXySrc_* / FUNCCODE_plcFrq0Src_* /
# FUNCCODE_plcRunMode_* / FUNCCODE_plcTimeUnit_* / FUNCCODE_motorCtrlMode_* 常量。
